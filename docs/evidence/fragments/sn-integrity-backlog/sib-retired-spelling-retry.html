UNWIND $cohort AS item
OPTIONAL MATCH (src:StandardNameSource {id: item.id})
OPTIONAL MATCH (candidate:StandardName {id: item.candidate})
OPTIONAL MATCH (candidate)-[:HAS_SUCCESSOR]->(successor:StandardName)
RETURN item.id AS source_id, item.candidate AS candidate_id,
       item.unit AS proposed_unit, src.id AS observed_source_id,
       src.source_id AS source_path, src.dd_unit AS snapshot_unit,
       src.last_error AS current_error, src.status AS source_status,
       candidate.id AS observed_candidate_id,
       candidate.name_stage AS candidate_stage, candidate.unit AS candidate_unit,
       candidate.description AS candidate_description,
       collect(DISTINCT successor {.id, .name_stage, .unit, .description,
                                 .reviewer_score_name}) AS successors
ORDER BY source_id

MATCH (n:StandardName)
RETURN count(n) AS names, count(n.id) AS ids,
       count(n.name_stage) AS stages, count(n.unit) AS units,
       COUNT { MATCH (:StandardName)-[:HAS_SUCCESSOR]->(:StandardName) } AS successor_edges

MATCH (n:StandardName)-[:HAS_SUCCESSOR]->(successor:StandardName)
RETURN n.id AS retired, n.name_stage AS retired_stage,
       successor.id AS successor, successor.name_stage AS successor_stage
ORDER BY n.id LIMIT 3
